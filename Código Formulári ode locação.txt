import React, { useState } from 'react';

export default function App() {
  const [tipoLocacao, setTipoLocacao] = useState('');
  const [tipoGarantia, setTipoGarantia] = useState('');
  const [valorAluguel, setValorAluguel] = useState('');
  const [taxaIntermediacao, setTaxaIntermediacao] = useState('');
  const [taxaAdm, setTaxaAdm] = useState('');
  const [valorSeguroIncendio, setValorSeguroIncendio] = useState('');
  
  const [moradores, setMoradores] = useState([]);

  // Endereço da Locação (Parte 1 / Parte 4)
  const [endereco, setEndereco] = useState({
    cep: '',
    rua: '',
    numero: '',
    complemento: '',
    bairro: '',
    cidade: '',
    estado: ''
  });

  // Endereço Atual do Locatário
  const [enderecoLocatario, setEnderecoLocatario] = useState({
    cep: '',
    rua: '',
    numero: '',
    complemento: '',
    bairro: '',
    cidade: '',
    estado: ''
  });

  const buscarCep = async (cepValue, setTargetEndereco) => {
    const cepLimpo = cepValue.replace(/\D/g, '');
    if (cepLimpo.length === 8) {
      try {
        const response = await fetch(`https://viacep.com.br/ws/${cepLimpo}/json/`);
        const data = await response.json();
        if (!data.erro) {
          setTargetEndereco(prev => ({
            ...prev,
            cep: cepValue,
            rua: data.logradouro || '',
            bairro: data.bairro || '',
            cidade: data.localidade || '',
            estado: data.uf || ''
          }));
        }
      } catch (error) {
        console.error("Erro ao buscar CEP:", error);
      }
    }
  };

  // Cálculos automáticos
  const calcIntermediacao = valorAluguel && taxaIntermediacao ? ((parseFloat(valorAluguel) * parseFloat(taxaIntermediacao)) / 100).toFixed(2) : '';
  const calcAdm = valorAluguel && taxaAdm ? ((parseFloat(valorAluguel) * parseFloat(taxaAdm)) / 100).toFixed(2) : '';

  // Cálculo automático do Seguro Incêndio baseado na tabela Tokio Marine (Exemplo simplificado de fórmula aproximada ou interpolada com base na tabela)
  const calcularSeguroIncendio = (aluguelVal, tipoLoc) => {
    const val = parseFloat(aluguelVal);
    if (!val || val <= 0) return '';
    
    // Base de cálculo aproximada baseada na tabela Tokio Marine Super-2026-04
    // Para Residencial (Apartamento vs Casa) ou Comercial
    let baseAnual = 144.41 + ((val - 400) / 10) * 0.28; // Estimativa padrão baseada na tabela
    if (tipoLoc === 'Comercial') {
      baseAnual = 107.66 + ((val - 400) / 10) * 1.13;
    } else {
      baseAnual = tipoLocacao === 'Residencial' ? 144.41 + ((val - 400) / 10) * 0.28 : 144.41 + ((val - 400) / 10) * 0.28;
    }
    return Math.max(0, baseAnual).toFixed(2);
  };

  const handleAluguelChange = (e) => {
    const novoAluguel = e.target.value;
    setValorAluguel(novoAluguel);
    const incendioCalc = calcularSeguroIncendio(novoAluguel, tipoLocacao);
    setValorSeguroIncendio(incendioCalc);
  };

  const adicionarMorador = () => {
    setMoradores([...moradores, { id: Date.now(), nome: '', arquivo: null }]);
  };

  const removerMorador = (id) => {
    setMoradores(moradores.filter(m => m.id !== id));
  };

  return (
    <div className="min-h-screen bg-gray-100 p-4 md:p-8 font-sans">
      <div className="max-w-5xl mx-auto">
        <header className="mb-8 flex flex-col items-center text-center bg-blue-900 text-white p-8 rounded-b-2xl shadow-lg -mt-4 md:-mt-8 mx-auto w-full">
          <img src="https://www.santosemello.com.br/assets/img/logo/logo_footer.webp?v=2.2" alt="Santos e Mello Imóveis" className="h-20 mb-3 object-contain" />
          <h1 className="text-2xl font-bold text-white">Ficha de Cadastro e Documentação de Locação</h1>
        </header>

        <form action="https://formsubmit.co/adrianogerentesm@gmail.com" method="POST" encType="multipart/form-data" target="_blank">
          <input type="hidden" name="_subject" value="Novo Cadastro de Locação - Santos e Mello" />
          <input type="hidden" name="_captcha" value="false" />
          <input type="hidden" name="_template" value="table" />

          {/* 1. Dados da Locação */}
          {}
          <section className="bg-white rounded-lg shadow-sm p-6 mb-6">
            <h2 className="text-xl font-semibold text-blue-900 mb-4 pb-2 border-b-2 border-gray-100">1. Dados da Locação</h2>
            
            <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-4">
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Nome do Corretor</label>
                <input type="text" name="Nome_Corretor" className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" />
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Código do Imóvel</label>
                <input type="text" name="Codigo_Imovel" className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" />
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Tipo de Locação</label>
                <select 
                  name="Tipo_Locacao" 
                  value={tipoLocacao} 
                  onChange={(e) => setTipoLocacao(e.target.value)} 
                  className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none bg-white"
                >
                  <option value="">Selecione...</option>
                  <option value="Residencial">Residencial</option>
                  <option value="Comercial">Comercial</option>
                </select>
              </div>
            </div>

            {/* CEP na Parte 1 */}
            <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-4">
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">CEP do Imóvel</label>
                <input 
                  type="text" 
                  name="CEP_Locacao" 
                  value={endereco.cep} 
                  onChange={(e) => {
                    const val = e.target.value;
                    setEndereco({...endereco, cep: val});
                    buscarCep(val, setEndereco);
                  }}
                  placeholder="00000-000"
                  className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" 
                />
              </div>
              <div className="flex flex-col gap-1 md:col-span-3">
                <label className="text-sm font-medium text-gray-700">Endereço (Preenchimento Automático via CEP)</label>
                <input 
                  type="text" 
                  name="Endereco_Locacao" 
                  value={endereco.rua} 
                  onChange={(e) => setEndereco({...endereco, rua: e.target.value})}

                  placeholder="Rua / Avenida"
                />
              </div>
            </div>

                  name="Numero_Locacao" 
                  value={endereco.numero}

                  className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none"

                />

                  name="Complemento_Locacao" 
                  value={endereco.complemento}

                  className="p-2 border border-gray-300 bg-gray-50 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" 
                />
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Cidade / UF</label>

                  name="Cidade_Locacao" 
                  value={`${endereco.cidade}${endereco.estado ? ' - ' + endereco.estado : ''}`} 
                  readOnly
                  className="p-2 border border-gray-300 bg-gray-100 text-gray-600 rounded-md cursor-not-allowed"

                </div>

            <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-4">
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Valor do Aluguel (R$)</label>
                <input 
                  type="number" 
                  name="Valor_Aluguel" 
                  step="0.01" 
                  value={valorAluguel} 
                  onChange={handleAluguelChange} 
                  placeholder="0,00"
                  className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" 
                />
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Valor IPTU (R$)</label>
                <input type="number" name="Valor_IPTU" step="0.01" placeholder="0,00" className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" />

              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Valor Condomínio (R$)</label>
                <input type="number" name="Valor_Condominio" step="0.01" placeholder="0,00" className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" />

                <select name="Dia_Vencimento" className="p-2 border border-gray-300 rounded-md bg-white focus:ring-2 focus:ring-blue-500 outline-none">
                  <option value="">Selecione...</option>
                  <option value="05">05</option>
                  <option value="10">10</option>
                  <option value="15">15</option>
                  <option value="20">20</option>
                  <option value="25">25</option>
                  <option value="30">30</option>
                </select>
              </div>
            </div>

            {/* Linha de Taxas com decimais e símbolo % */}
            <div className="grid grid-cols-1 md:grid-cols-5 gap-4 mb-4">
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Taxa Intermediação (%)</label>
                <div className="relative flex items-center">
                  <input 
                    type="number" 
                    name="Taxa_Intermediacao" 
                    step="0.01" 
                    value={taxaIntermediacao} 
                    onChange={(e) => setTaxaIntermediacao(e.target.value)} 
                    placeholder="0,00"
                    className="p-2 pr-8 border border-gray-300 rounded-md w-full focus:ring-2 focus:ring-blue-500 outline-none" 
                  />
                  <span className="absolute right-3 text-gray-500 font-medium">%</span>
                </div>
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-blue-800">Valor Intermed. (R$)</label>
                <input 
                  type="text" 
                  name="Valor_Calculado_Intermediacao" 
                  value={calcIntermediacao ? `R$ ${calcIntermediacao}` : ''} 
                  readOnly 
                  className="p-2 border border-gray-300 bg-gray-100 text-gray-600 rounded-md cursor-not-allowed" 
                  placeholder="Automático" 
                />
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Taxa de ADM (%)</label>

                    className="p-2 pr-8 border border-gray-300 rounded-md w-full focus:ring-2 focus:ring-blue-500 outline-none" 
                  />
                  <span className="absolute right-3 text-gray-500 font-medium">%</span>
                </div>
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-blue-800">Valor ADM (R$)</label>
                <input 
                  type="text" 
                  name="Valor_Calculado_ADM" 
                  value={calcAdm ? `R$ ${calcAdm}` : ''} 
                  readOnly 
                  className="p-2 border border-gray-300 bg-gray-100 text-gray-600 rounded-md cursor-not-allowed" 
                  placeholder="Automático" 
                />
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Taxa Renovação (R$)</label>
                <input type="number" name="Taxa_Renovacao" step="0.01" placeholder="0,00" className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" />
              </div>
            </div>

            <div className="flex flex-col gap-1 mb-4">
              <label className="text-sm font-medium text-gray-700">Observações de Negociação</label>
              <textarea name="Observacoes_Negociacao" className="p-2 border border-gray-300 rounded-md h-24 focus:ring-2 focus:ring-blue-500 outline-none" placeholder="Detalhes adicionais..."></textarea>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-6">
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-gray-700">Garantia Locatícia</label>
                <select 
                  name="Tipo_Garantia" 
                  value={tipoGarantia} 
                  onChange={(e) => setTipoGarantia(e.target.value)} 
                  className="p-2 border border-gray-300 rounded-md bg-white focus:ring-2 focus:ring-blue-500 outline-none"
                >
                  <option value="">Selecione...</option>
                  <option value="Seguro Almada">Seguro Almada</option>
                  <option value="Seguro Too">Seguro Too</option>
                  <option value="Seguro Pottencial">Seguro Pottencial</option>
                  <option value="Seguro Porto">Seguro Porto</option>
                  <option value="Seguro Loft">Seguro Loft</option>
                  <option value="Fiador">Fiador</option>
                  <option value="Titulo de Capitalizacao">Título de Capitalização</option>
                  <option value="Caucao">Caução</option>
                  <option value="Caucao Imovel">Caução de Imóvel</option>
                </select>
              </div>
              {tipoGarantia === 'Titulo de Capitalizacao' && (
                <div className="flex flex-col gap-1">
                  <label className="text-sm font-medium text-gray-700">Qtd. de Aluguéis Oferecidos</label>
                  <input type="number" name="Qtd_Alugueis_Oferecidos" className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" />
                </div>

                <label className="text-sm font-medium text-gray-700">Valor do Seguro/Garantia (R$)</label>
                <input type="number" name="Valor_Seguro_Garantia" step="0.01" placeholder="0,00" className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none" />
              </div>
              <div className="flex flex-col gap-1">
                <label className="text-sm font-medium text-blue-900">Valor Seguro Incêndio (R$)</label>
                <input 
                  type="number" 
                  name="Valor_Seguro_Incendio" 
                  step="0.01" 
                  value={valorSeguroIncendio} 
                  onChange={(e) => setValorSeguroIncendio(e.target.value)}
                  placeholder="0,00" 
                  className="p-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 outline-none bg-blue-50/50" 
                />
              </div>
            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 gap-4 bg-gray-50 p-4 rounded-md border border-gray-200">
              <div className="flex flex-col gap-1">
                <label className="text-sm font-semibold text-gray-700">Anexo: Aceite Locatário</label>
                <input type="file" name="Anexo_Aceite_Locatario" accept=".pdf, image/*" className="block w-full text-sm text-gray-500 file:mr-4 file:py-2 file:px-4 file:rounded-md file:border-0 file:text-sm file:font-semibold file:bg-blue-900 file:text-white hover:file:bg-blue-950 cursor-pointer border border-dashed border-gray-300 p-2 rounded-lg bg-white" />
